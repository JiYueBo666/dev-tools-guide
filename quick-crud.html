2. 接通任务接口

这一节新增四个文件。四个都保存后，再重启服务。

目录固定为 src/main/java/com/example/quick。Controller 接请求，Service 办业务，Repository 访问数据库，Entity 对应表。以后主要改 Controller 和 Service。

TaskEntity.java

文件：src/main/java/com/example/quick/TaskEntity.java

怎么用：固定保留 Entity、Id、编号生成和无参数构造方法。要新增字段，再修改这里及对应请求/响应字段。表名使用 quick_tasks，避免与旧项目的 tasks 混淆。

package com.example.quick;
import jakarta.persistence.*;
@Entity
@Table(name="quick_tasks")
public class TaskEntity {
    @Id @GeneratedValue(strategy=GenerationType.IDENTITY)
    private Long id;
    @Column(nullable=false, length=200)
    private String content;
    private boolean done;
    private boolean deleted;
    protected TaskEntity() {}
    public TaskEntity(String content) { this.content = content; }
    public Long getId() { return id; }
    public String getContent() { return content; }
    public boolean isDone() { return done; }
    public boolean isDeleted() { return deleted; }
    public void setDone(boolean done) { this.done = done; }
    public void markDeleted() { deleted = true; }
}

TaskRepository.java

文件：src/main/java/com/example/quick/TaskRepository.java

怎么用：基础增删改查从 JpaRepository 继承。按条件查询的方法写在这里，方法名中的字段必须与实体字段一致。

package com.example.quick;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import java.util.Optional;
public interface TaskRepository extends JpaRepository<TaskEntity, Long> {
    Page<TaskEntity> findByDeletedFalse(Pageable pageable);
    Page<TaskEntity> findByDeletedFalseAndDone(boolean done, Pageable pageable);
    Optional<TaskEntity> findByIdAndDeletedFalse(long id);
}

TaskService.java

文件：src/main/java/com/example/quick/TaskService.java

怎么用：保留事务注解、构造方法和 required。改业务规则时主要修改各方法内部；TaskView 是接口输出，TaskPage 是分页输出。

package com.example.quick;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import java.util.List;
@Service
public class TaskService {
    public record TaskView(long id, String content, boolean done) {
        static TaskView of(TaskEntity t) { return new TaskView(t.getId(), t.getContent(), t.isDone()); }
    }
    public record TaskPage(List<TaskView> items, long total, int page, int size) {}
    public static class Missing extends RuntimeException {}
    private final TaskRepository repository;
    public TaskService(TaskRepository repository) { this.repository = repository; }
    @Transactional
    public TaskView create(String content) { return TaskView.of(repository.save(new TaskEntity(content.strip()))); }
    @Transactional(readOnly=true)
    public TaskView find(long id) { return TaskView.of(required(id)); }
    @Transactional(readOnly=true)
    public TaskPage list(int page, int size, Boolean done) {
        // 对外页码从 1 开始，排序字段固定，避免直接接受任意列名。
        var request = PageRequest.of(page - 1, size, Sort.by("id").descending());
        var result = done == null ? repository.findByDeletedFalse(request)
                : repository.findByDeletedFalseAndDone(done, request);
        return new TaskPage(result.map(TaskView::of).getContent(), result.getTotalElements(), page, size);
    }
    @Transactional
    public TaskView changeDone(long id, boolean done) {
        var task = required(id);
        task.setDone(done); // false 也会写回；方法成功返回后提交事务。
        return TaskView.of(task);
    }
    @Transactional
    public void delete(long id) { required(id).markDeleted(); }
    @Transactional
    public List<TaskView> importTasks(List<String> contents) {
        var entities = contents.stream().map(String::strip).map(TaskEntity::new).toList();
        return repository.saveAll(entities).stream().map(TaskView::of).toList();
    }
    private TaskEntity required(long id) {
        return repository.findByIdAndDeletedFalse(id).orElseThrow(Missing::new);
    }
}

TaskController.java

文件：src/main/java/com/example/quick/TaskController.java

怎么用：类上的 /api/tasks 是共同前缀。方法上的路径是子路径。路径变量用 @PathVariable；问号后的参数用 @RequestParam；JSON 正文用 @RequestBody；要校验时加 @Valid。

package com.example.quick;
import jakarta.validation.Valid;
import jakarta.validation.constraints.*;
import org.springframework.web.bind.annotation.*;
import org.springframework.http.ResponseEntity;
import java.net.URI;
import java.util.List;
@RestController
@RequestMapping("/api/tasks")
public class TaskController {
    public record Create(@NotBlank @Size(max=200) String content) {}
    public record Change(@NotNull Boolean done) {}
    public record Import(@NotEmpty @Size(max=100) List<@NotBlank @Size(max=200) String> contents) {}
    private final TaskService service;
    public TaskController(TaskService service) { this.service = service; }
    @PostMapping
    public ResponseEntity<TaskService.TaskView> create(@Valid @RequestBody Create input) {
        var result = service.create(input.content());
        return ResponseEntity.created(URI.create("/api/tasks/" + result.id())).body(result);
    }
    @GetMapping("/{id}")
    public TaskService.TaskView find(@PathVariable @Positive long id) { return service.find(id); }
    @GetMapping
    public TaskService.TaskPage list(@RequestParam(defaultValue="1") @Min(1) int page,
            @RequestParam(defaultValue="20") @Min(1) @Max(100) int size,
            @RequestParam(required=false) Boolean done) {
        return service.list(page, size, done);
    }
    @PatchMapping("/{id}")
    public TaskService.TaskView change(@PathVariable @Positive long id, @Valid @RequestBody Change input) {
        return service.changeDone(id, input.done());
    }
    @DeleteMapping("/{id}")
    public ResponseEntity<Void> delete(@PathVariable @Positive long id) {
        service.delete(id);
        return ResponseEntity.noContent().build();
    }
    @PostMapping("/imports")
    public List<TaskService.TaskView> importTasks(@Valid @RequestBody Import input) {
        return service.importTasks(input.contents());
    }
}
